## 1. Цели и задачи

### 1.1 Бизнес-цель
Реализовать функционал управления портфолио в админ-панели для демонстрации выполненных работ на публичном сайте, что повысит доверие потенциальных клиентов и увеличит конверсию.

### 1.2 Пользовательская ценность
- Администратор может добавлять, редактировать и удалять работы портфолио через удобный интерфейс
- Посетители сайта видят актуальные примеры выполненных работ с фотографиями
- Возможность загрузки до 5 фотографий для каждой работы

### 1.3 Метрики успеха
- Администратор может создать элемент портфолио за < 2 минут
- Загрузка изображений работает корректно (валидация, превью, автоматическая генерация миниатюр)
- Публичная страница портфолио отображает данные из БД
- Существующий дизайн страницы портфолио сохранён
- Drag & drop сортировка работает плавно (время отклика < 300ms)
- Массовые операции выполняются корректно (до 50 элементов за раз)

## 3. Нефункциональные требования

### 3.1 Производительность
- Время загрузки страницы портфолио < 1 сек
- Оптимизация изображений: автоматическая генерация миниатюр (400x300px) при загрузке
- Ленивая загрузка изображений на публичном сайте
- Кеширование списка портфолио на публичном API (5 минут)
- Drag & drop операции < 300ms
- Bulk операции оптимизированы (batch updates в БД)

### 3.2 Безопасность

#### 3.2.1 Аутентификация и авторизация
- Все административные endpoints требуют аутентификации
- Доступ только для ролей: ADMIN, CONTENT_MANAGER
- Проверка сессии на каждом запросе

#### 3.2.2 Загрузка файлов
- Валидация MIME-типа на сервере (не только расширение!)
- Проверка magic bytes файла
- Санитизация имени файла: только латинские буквы, цифры, дефис, подчёркивание
- Генерация уникального имени файла (UUID + оригинальное расширение)
- Ограничение размера файла: 5 MB
- Проверка реальных размеров изображения (минимум 200x200px)
- Удаление EXIF-данных из изображений (privacy)
- **Автоматическая генерация миниатюр** (400x300px, cover mode)
- Сохранение оригинала + миниатюры с суффиксом `_thumb`

```typescript
const ALLOWED_MIME_TYPES = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
const MAX_FILE_SIZE = 5 * 1024 * 1024; // 5 MB
const MIN_IMAGE_DIMENSIONS = { width: 200, height: 200 };
```

#### 3.2.3 Защита от атак
- CSRF защита на всех mutating запросах
- Rate limiting на upload endpoint: 10 запросов/мин
- Валидация всех входных данных (Zod схемы)
- Логирование всех CRUD операций в AuditLog

### 3.3 Масштабируемость
- Хранение файлов в файловой системе (public/uploads/portfolio/)
- Структура папок: `public/uploads/portfolio/{year}/{month}/{uuid}.{ext}`
- Возможность миграции на S3 в будущем без изменения API

### 4.2 API спецификация

#### GET /api/admin/portfolio
**Auth**: Bearer JWT (role: ADMIN, CONTENT_MANAGER)

**Query Parameters:**
- `page` (optional, default: 1) - номер страницы
- `pageSize` (optional, default: 20) - элементов на странице
- `search` (optional) - поиск по названию
- `category` (optional) - фильтр по категории (fence/canopy)
- `active` (optional) - фильтр по активности (true/false)

**Response 200:**
```json

      "title": "Забор из профнастила",
      "category": "fence",
```

```typescript
export const portfolioInputSchema = z.object({
  title: z.string().min(1, 'Название обязательно').max(255, 'Максимум 255 символов'),
  category: z.enum(['fence', 'canopy'], {
    errorMap: () => ({ message: 'Выберите категорию' }),

  type: z.string().max(100, 'Максимум 100 символов').optional(),
  description: z.string().max(2000, 'Максимум 2000 символов').optional(),
  images: z.array(z.string().url())
    .min(1, 'Минимум 1 изображение')
    .max(5, 'Максимум 5 изображений'),
  cost: z.number().min(0, 'Стоимость не может быть отрицательной').optional(),
  showCost: z.boolean().optional().default(false),
  active: z.boolean().optional().default(true),
});

export const portfolioUpdateSchema = portfolioInputSchema.partial();
```

```typescript
// src/services/admin/portfolioService.ts
import { prisma } from '@/lib/prisma';
import { PortfolioInput, PortfolioUpdate, PortfolioListParams } from '@/lib/validators/portfolio';
import { deleteImage } from '@/lib/utils/fileUpload';

export class PortfolioService {
  async getAll(params: PortfolioListParams) {
    const { page = 1, pageSize = 20, search, category, active } = params;
    const skip = (page - 1) * pageSize;

    const where: any = {};

    if (search) {
      where.OR = [
        { title: { contains: search, mode: 'insensitive' } },
        { description: { contains: search, mode: 'insensitive' } },
      ];
    }

    if (category) {
      where.category = category;
    }

    if (active !== undefined) {
      where.active = active;
    }

    const [items, total] = await Promise.all([
      prisma.portfolioItem.findMany({
        where,
        skip,
        take: pageSize,
        orderBy: [{ sortOrder: 'asc' }, { createdAt: 'desc' }],
      }),
      prisma.portfolioItem.count({ where }),
    ]);

    return {
      items,
      total,
      page,
      pageSize,
      totalPages: Math.ceil(total / pageSize),
    };
  }

  async create(data: PortfolioInput, userId: string) {
    const maxSortOrder = await prisma.portfolioItem.aggregate({
      _max: { sortOrder: true },
    });

    const item = await prisma.portfolioItem.create({
      data: {
        ...data,
        sortOrder: (maxSortOrder._max.sortOrder || 0) + 1,
      },
    });

    await this.logAction(item.id, 'CREATE', null, item, userId);

    return item;
  }

  async update(id: string, data: PortfolioUpdate, userId: string) {
    const oldItem = await prisma.portfolioItem.findUnique({ where: { id } });
    
    if (!oldItem) {
      throw new Error('Элемент портфолио не найден');
    }

    // Удалить старые изображения, которые были удалены
    if (data.images) {
      const oldImages = oldItem.images as string[];
      const newImages = data.images;
      const removedImages = oldImages.filter(img => !newImages.includes(img));
      
      for (const img of removedImages) {
        await deleteImage(img);
      }
    }

    const item = await prisma.portfolioItem.update({
      where: { id },
      data,
    });

    await this.logAction(id, 'UPDATE', oldItem, item, userId);

    return item;
  }

  async delete(id: string, userId: string) {
    const item = await prisma.portfolioItem.findUnique({ where: { id } });
    
    if (!item) {
      throw new Error('Элемент портфолио не найден');
    }

    // Удалить все изображения
    const images = item.images as string[];
    for (const img of images) {
      await deleteImage(img);
    }

    await prisma.portfolioItem.delete({ where: { id } });

    await this.logAction(id, 'DELETE', item, null, userId);
  }

  async toggleActive(id: string, userId: string) {
    const item = await prisma.portfolioItem.findUnique({ where: { id } });
    
    if (!item) {
      throw new Error('Элемент портфолио не найден');
    }

    const updated = await prisma.portfolioItem.update({
      where: { id },
      data: { active: !item.active },
    });

    await this.logAction(id, 'TOGGLE_ACTIVE', item, updated, userId);

    return updated;
  }

  private async logAction(entityId: string, action: string, oldValues: any, newValues: any, userId: string) {
    await prisma.auditLog.create({
      data: {
        userId,
        action: `PORTFOLIO_${action}`,
        entityType: 'PortfolioItem',
        entityId,
        oldValues,
        newValues,
      },
    });
  }
}

export const portfolioService = new PortfolioService();
```

### A.2 File Upload Utilities

```typescript
// src/lib/utils/fileUpload.ts
import { writeFile, unlink, mkdir } from 'fs/promises';
import { existsSync } from 'fs';
import path from 'path';
import { v4 as uuidv4 } from 'uuid';
import sharp from 'sharp';

const ALLOWED_MIME_TYPES = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
const MAX_FILE_SIZE = 5 * 1024 * 1024; // 5 MB
const MIN_DIMENSIONS = { width: 200, height: 200 };

const MAGIC_BYTES: Record<string, number[]> = {
  'image/jpeg': [0xff, 0xd8, 0xff],
  'image/png': [0x89, 0x50, 0x4e, 0x47],
  'image/webp': [0x52, 0x49, 0x46, 0x46],
  'image/gif': [0x47, 0x49, 0x46, 0x38],
};

export function validateImageFile(file: File): { valid: boolean; error?: string } {
  if (!ALLOWED_MIME_TYPES.includes(file.type)) {
    return { valid: false, error: 'Поддерживаются только форматы: JPEG, PNG, WebP, GIF' };
  }

  if (file.size > MAX_FILE_SIZE) {
    return { valid: false, error: 'Размер файла не должен превышать 5 MB' };
  }

  return { valid: true };
}

export function sanitizeFilename(filename: string): string {
  const ext = path.extname(filename);
  const name = path.basename(filename, ext);
  const sanitized = name
    .replace(/[^a-zA-Z0-9а-яА-Я_-]/g, '-')
    .replace(/-+/g, '-')
    .substring(0, 50);
  return `${sanitized}${ext}`;
}

export function generateUniqueFilename(ext: string): string {
  return `${uuidv4()}${ext}`;
}

export function getUploadPath(): string {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  return path.join(process.cwd(), 'public', 'uploads', 'portfolio', String(year), month);
}

export async function saveImage(file: File): Promise<string> {
  const buffer = Buffer.from(await file.arrayBuffer());
  
  // Validate magic bytes
  const mimeType = detectMimeType(buffer);
  if (!mimeType || !ALLOWED_MIME_TYPES.includes(mimeType)) {
    throw new Error('Invalid file type');
  }

  // Check dimensions
  const metadata = await sharp(buffer).metadata();
  if (metadata.width! < MIN_DIMENSIONS.width || metadata.height! < MIN_DIMENSIONS.height) {
    throw new Error(`Minimum dimensions: ${MIN_DIMENSIONS.width}x${MIN_DIMENSIONS.height}`);
  }

  // Generate paths
  const ext = path.extname(file.name) || `.${metadata.format}`;
  const filename = generateUniqueFilename(ext);
  const uploadPath = getUploadPath();
  const filePath = path.join(uploadPath, filename);

  // Ensure directory exists
  if (!existsSync(uploadPath)) {
    await mkdir(uploadPath, { recursive: true });
  }

  // Strip EXIF and save original
  await sharp(buffer)
    .rotate() // Auto-orient based on EXIF
    .withMetadata({ exif: undefined }) // Remove EXIF
    .toFile(filePath);

  // Generate thumbnail (400x300px, cover mode)
  const thumbFilename = filename.replace(/(\.\w+)$/, '_thumb$1');
  const thumbPath = path.join(uploadPath, thumbFilename);
  
  await sharp(buffer)
    .rotate()
    .resize(400, 300, { fit: 'cover', position: 'center' })
    .withMetadata({ exif: undefined })
    .toFile(thumbPath);

  // Return public URL (original)
  const publicUrl = filePath.replace(path.join(process.cwd(), 'public'), '');
  return publicUrl;
}

export async function deleteImage(url: string): Promise<void> {
  const filePath = path.join(process.cwd(), 'public', url);
  const thumbPath = filePath.replace(/(\.\w+)$/, '_thumb$1');
  
  // Delete original
  if (existsSync(filePath)) {
    await unlink(filePath);
  }
  
  // Delete thumbnail
  if (existsSync(thumbPath)) {
    await unlink(thumbPath);
  }
}

function detectMimeType(buffer: Buffer): string | null {
  for (const [mimeType, bytes] of Object.entries(MAGIC_BYTES)) {
    const matches = bytes.every((byte, index) => buffer[index] === byte);
    if (matches) return mimeType;
  }
  return null;
}
```
